Private Function WKDAY(DATEJ) As String ''********************SUBROUTINE TO CALCULATE UTC DAY OF THE WEEK**********
 WKDAY = 1 + DATEJ - Int(DATEJ / 7) * 7
 On WKDAY GoTo 2995, 2997, 2999, 3001, 3003, 3005, 3007
2995 WKDAY = "SUNDAY": GoTo 3008
2997 WKDAY = "MONDAY": GoTo 3008
2999 WKDAY = "TUESDAY": GoTo 3008
3001 WKDAY = "WEDNESDAY": GoTo 3008
3003 WKDAY = "THURSDAY": GoTo 3008
3005 WKDAY = "FRIDAY": GoTo 3008
3007 WKDAY = "SATURDAY"
3008 'Continue
End Function

Private Function FNA(X) As Single 'converts radians to degrees and rounds to nearest tenth
FNA = Int(X * (360 / (3.141592656 * 2)) * 10 + 0.5) / 10

End Function
 
Private Function FNB(X) As Single
FNB = Int(X * 100 + 0.5) / 100
End Function

Private Function FNC(X) As Double

FNC = (X - Int(X)) * (3.141592656 * 2)
End Function

Private Function FNR(A, B) As Double
FNR = Int(A * B + 0.5) / B
End Function
Private Function FNL(X) As Double ' COMMON LOGARITHM OF X
FNL = Log(X) / Log(10)
End Function
 
Private Function FNJULIAN(AY, AM, AD) As Single 'JULIAN DATE (MINUS 2397547.5) FOR 0000 HOURS UTC, YEARS 1900-2099
FNJULIAN = 367 * AY - Int(7 * (AY + Int((AM + 9) / 12)) / 4) + Int(275 * AM / 9) + AD - 676534!
End Function
 
Private Function FNATAN2(S, C) As Double
FNATAN2 = Atn(S / C) - Sgn(S) * 3.141592656 * (Fix(Sgn(C) - 0.5))
 End Function
 
Private Function FNPOL(AZ, EL, LAT) As Single 'CROSS POLARIZATION ANGLE WITH RESPECT TO EARTH AXIS IN DEGREES
TUPI = 2 * 3.141592656
Deg = 360 / TUPI
FNPOL = Int(Atn((Sin(LAT) * Cos(EL) - Cos(LAT) * Cos(AZ) * Sin(EL)) / (Cos(LAT) * Sin(AZ))) * Deg)

End Function

Private Sub Topo(ULATD, RA, GHA, DEC, RAT, GHAT, DECT, RANGE, GAST)
  Dim gaux As Double
  Dim mpar As Double
  Dim gclat As Double
  Dim gdist As Double '  (RHO)
    
  '  2010-04-04 ==========Topocentric calcs for Moon RA DEC GHA ===============================
  '  Details from http:' stjarnhimlen.se/comp/tutorial.html#9
  '  We are going to calculate station (Topocentric) Moon RA and Declination (useful for occultations and for Polar mount Trackers)
  
  '  Apparent size of the equatorial radius of the Earth, as seen from the Moon.
  ' mpar = Atn(Sin(6378.16 / RANGE) / Cos(Sqr(RANGE * RANGE - 6378.16 * 6378.16)) / RANGE)
  mpar = FNATAN2(Sin(6378.16 / RANGE), Cos(Sqr(RANGE * RANGE - 6378.16 * 6378.16) / RANGE))
  
  '  Convert Astronomical Latitude into Geocentric Latitude
  gclat = ULATD - (0.1924 * TUPI / 360) * Sin(2 * ULATD) ' Geocentric Latitude of station in radians
  gdist = 0.99833 + 0.00167 * Cos(2 * ULATD) ' Station distance from centre of Earth in Equatorial Earth Radii ("rho")
  
  ' Auxiliary angle
  
  gaux = Atn(Tan(gclat) / Cos(GHA)) ' GHA used for HA
  
  '  Calculate Topocentric Moon RA
  RAT = RA - mpar * gdist * Cos(gclat) * Sin(GHA) / Cos(DEC)
    
  ' Calculate Topocentric Moon Declination.
  DECT = DEC - mpar * gdist * Sin(gclat) * Sin(gaux - DEC) / Sin(gaux) ' DEC is Moon's geocentric Dec in radians
    
  ' Calculate Moon topocentric GHA (L is GST)
  GHAT = GAST / 24 * TUPI - RAT       ' GHA of Moon radians (topocentric)
  
  If GHAT < TUPI And GHAT < 0 Then
    GHAT = GHAT + TUPI
  End If
  
  If GHAT >= TUPI Then
    GHAT = GHAT - TUPI
  End If

 End Sub

1 Dim CALLSIGN As String: txtSearch.Text = ""

If Left(txtDateCalc.Text, 5) = "Exact" Then GoTo 2 Else GoTo 3

2 If KeyAscii = 91 Then
    Let txtSearch.Text = "[[[[["
    Call cmdDXLookup_Click
    KeyAscii = 0
    GoTo 6
End If
If KeyAscii = 93 Then
    Let txtSearch.Text = "]]]]]"
    Call cmdDXLookup_Click
    KeyAscii = 0
    GoTo 6
End If

3 If KeyAscii = 91 Or KeyAscii = 93 Then Let KeyAscii = 0

If KeyAscii = 8 Then txtCallsignDX.Text = "" 'clear field on Backspace
If KeyAscii > 122 Then txtCallsigDX.Text = ""

CALLSIGN = txtCallsignDX.Text

Let I = Len(CALLSIGN)
If I = 12 Then Let KeyAscii = 0
    Let N = KeyAscii
 If KeyAscii = 42 Then GoTo 4
 If KeyAscii < 47 Then GoTo 5
 If KeyAscii < 58 Then GoTo 4
 
    If N > 96 Then Let N = N - 32
4   txtCallsignDX.Text = Left(CALLSIGN, I) + Chr(N)
    KeyAscii = 10

5 ' Continue
If KeyAscii = 27 Then Call cmdDXLookup_Click
6 'continue
txtCallsignDX.SetFocus

End Sub

Private Sub txtYearStart_Click()
'For I = 1 To 2
AM = cboMonthStart.ListIndex + 1
If cboDayStart.ListIndex > -1 Then Let AD = cboDayStart.ListIndex + 1 Else Let AD = Val(cboDayStart.Text)
AY = Val(txtYearStart.Text)
On (AM) GoTo 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31
28 If AD > 28 Then GoTo 29 Else GoTo 31
29 Let Max = 28
If (AY Mod 4) = 0 Then Let Max = 29
If (AY Mod 100) = 0 Then Let Max = 28
If (AY Mod 400 = 0) And (AY Mod 100 = 0) Then Max = 29
If AD < Max Then GoTo 31 Else Let AD = Max
GoTo 31
30 If AD > 30 Then Let AD = 30: GoTo 31
31 'Continue
Let cboDayStart.Text = AD
Let txtYearStart.Text = Str(AY)
startdate = DateSerial(AY, AM, AD)
startdate = DateValue(Str(startdate))
Let cboMonthEnd.ListIndex = cboMonthStart.ListIndex
If cboDayStart.ListIndex > -1 Then Let cboDayEnd.ListIndex = cboDayStart.ListIndex
Let txtYearEnd.Text = txtYearStart.Text
'Next I
Call cboMonthStart_Click
End Sub
